CREATE TABLE utilisateur(
   id_utilisateur SERIAL,
   nom_utilisateur VARCHAR(50)  NOT NULL,
   email VARCHAR(50)  NOT NULL,
   mot_de_passe VARCHAR(255)  NOT NULL,
   token VARCHAR(255)  NOT NULL,
   date_creation_compte TIMESTAMP NOT NULL,
   PRIMARY KEY(id_utilisateur),
   UNIQUE(nom_utilisateur),
   UNIQUE(email)
);

CREATE TABLE cryptomonnaie(
   id_cryptomonnaie SERIAL,
   nom VARCHAR(50)  NOT NULL,
   symbole VARCHAR(10) ,
   PRIMARY KEY(id_cryptomonnaie),
   UNIQUE(nom),
   UNIQUE(symbole)
);

CREATE TABLE portefeuille(
   id_portefeuille SERIAL,
   montant NUMERIC(15,8)  ,
   id_cryptomonnaie_cryptomonnaie INTEGER NOT NULL,
   id_utilisateur_utilisateur INTEGER NOT NULL,
   PRIMARY KEY(id_portefeuille),
   FOREIGN KEY(id_cryptomonnaie_cryptomonnaie) REFERENCES cryptomonnaie(id_cryptomonnaie),
   FOREIGN KEY(id_utilisateur_utilisateur) REFERENCES utilisateur(id_utilisateur)
);

CREATE TABLE transaction(
   id_transaction SERIAL,
   montant NUMERIC(15,8)  ,
   est_valide BOOLEAN,
   date_transaction TIMESTAMP,
   id_utilisateur_vendeur INTEGER NOT NULL,
   id_utilisateur_acheteur INTEGER NOT NULL,
   PRIMARY KEY(id_transaction),
   FOREIGN KEY(id_utilisateur_vendeur) REFERENCES utilisateur(id_utilisateur),
   FOREIGN KEY(id_utilisateur_acheteur) REFERENCES utilisateur(id_utilisateur)
);

CREATE TABLE mvt_fond(
   id_mvt_fond SERIAL,
   depot NUMERIC(15,8)  ,
   retrait NUMERIC(15,8)  ,
   date_mvt TIMESTAMP,
   id_utilisateur_utilisateur INTEGER NOT NULL,
   PRIMARY KEY(id_mvt_fond),
   FOREIGN KEY(id_utilisateur_utilisateur) REFERENCES utilisateur(id_utilisateur)
);

CREATE TABLE cours_crypto(
   id_cours_crypto SERIAL,
   date_cours TIMESTAMP NOT NULL,
   montant INTEGER NOT NULL,
   id_cryptomonnaie INTEGER NOT NULL,
   PRIMARY KEY(id_cours_crypto),
   UNIQUE(date_cours),
   FOREIGN KEY(id_cryptomonnaie) REFERENCES cryptomonnaie(id_cryptomonnaie)
);

CREATE TABLE validation_mvt(
   id_validation SERIAL,
   date_validation DATE NOT NULL,
   id_utilisateur_utilisateur INTEGER NOT NULL,
   id_mvt_fond_mvt_fond INTEGER NOT NULL,
   PRIMARY KEY(id_validation),
   FOREIGN KEY(id_utilisateur_utilisateur) REFERENCES utilisateur(id_utilisateur),
   FOREIGN KEY(id_mvt_fond_mvt_fond) REFERENCES mvt_fond(id_mvt_fond)
);
